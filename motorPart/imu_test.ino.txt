#include <Wire.h>
#include <MPU6050_light.h> 

MPU6050 mpu(Wire);
unsigned long timer = 0;

void setup() {
  Serial.begin(115200);
  Wire.begin(); // I2C 통신 시작

  // --- 1. 센서 연결 확인 ---
  byte status = mpu.begin();
  
  if (status != 0) {
    while(1); 
  }

  delay(1000);
  mpu.calcOffsets(); 
}

void loop() {
  mpu.update();

  if(millis() - timer > 50) { 
    
    // Theta 값 얻기 
    float heading = mpu.getAngleZ();
    Serial.print("Heading: ");
    Serial.println(heading);

    timer = millis();
  }
}
